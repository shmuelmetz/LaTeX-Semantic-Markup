% Manual smoke test for shmuelsemtex.sty -- exercises every macro
% currently defined in shmuelsemtex.dtx. Not a CTAN test suite;
% just a quick "does it still compile and render sanely" check.
% Requires shmuelsemtex.sty on TEXINPUTS (e.g. run
% `tex shmuelsemtex.ins` in this directory first).
\documentclass{article}
\usepackage{amsmath}
\usepackage{shmuelsemtex}
\begin{document}
Category name: $\catName{A}$.

Category name, long/conventional (same macro, auto-detected
typeface -- author never chooses): $\catName{Set}$, $\catName{Top}$.

Sequence of categories: $\catSeqName{A}$.

(They render differently: $\catName{A}$ vs.\ $\catSeqName{A}$, and are
different macros, not the same macro with a flag.)

Object name: $\objName{x}$.

Morphism: $\morphName{f} \colon \objName{x} \morphMono \objName{y}$,
and $\objName{x} \morphEpi \objName{y}$.

Sequence of morphisms (distinct from a single morphism):
$\morphName{f}$ vs.\ $\morphSeqName{f}$.

Composition: $\morphName{g} \morphCompose \morphName{f}$ and
$\morphName{g} \morphCompose[\text{def}] \morphName{f}$.

Head/tail composition (distinct operations, not styling variants):
$\morphName{g} \morphComposeHead \morphName{f}$ vs.\
$\morphName{g} \morphComposeTail \morphName{f}$.

Chart/atlas/coordinate: $\chartName{U}$, $\atlasName{\mathcal{A}}$,
$\coordName{x}$.

Generic: \semName{plain text}.
\end{document}
